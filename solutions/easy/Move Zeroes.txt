// Title: Move Zeroes
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/move-zeroes/

class Solution:
    def moveZeroes(self, nums: list[int]) -> None:
        """
        Do not return anything, modify nums in-place instead.
        """
        result = []
        count = 0
        for i in range(len(nums)):
            if nums[i] ==0:
                count +=1
            else:
                result.append(nums[i])
        for i in range(count):
            result.append(0)
        nums[:] = result
        return nums
