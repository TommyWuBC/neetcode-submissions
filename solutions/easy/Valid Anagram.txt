// Title: Valid Anagram
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/valid-anagram/

class Solution:
    def isAnagram(self, s: str, t: str) -> bool:
        map1 = {}
        map2 = {}
        for char in s:
            map1[char] = map1.get(char, 0) +1
        for char in t:
            map2[char] = map2.get(char, 0)+1
        return map2 == map1
