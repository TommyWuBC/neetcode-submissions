// Title: 3Sum
            // Difficulty: Medium
            // Language: Python3
            // Link: https://leetcode.com/problems/3sum/

class Solution:
    def threeSum(self, nums: list[int]) -> list[list[int]]:
        nums.sort()
        result =[]
        seen = set()
        for i in range(len(nums)):
            if i > 0 and nums[i] == nums[i-1]:
                continue
            map = {}
            target = 0-nums[i]
            for j in range(i+1, len(nums)):
                jtarget = target - nums[j]
                if jtarget in map and map[jtarget] != j:
                    tup = tuple(sorted([nums[i], nums[j], jtarget]))
                    if tup not in seen:
                        result.append([nums[i], nums[j], jtarget])
                        seen.add(tup)
                map[nums[j]] = j
        return result
