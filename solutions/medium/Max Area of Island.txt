class Solution:
    def maxAreaOfIsland(self, grid: list[list[int]]) -> int:
        self.grid = grid
        maximum = 0
        for i in range(len(grid)):
            for j in range(len(grid[0])):
                if grid[i][j] ==1:
                    area = 0
                    self.area = area
                    self.dfs(i, j)
                    if self.area>maximum:
                        maximum = self.area
        return maximum
        
    def dfs(self, i, j):
        if i <0 or i >= len(self.grid) or j< 0 or j>=len(self.grid[0]):
            return
        if self.grid[i][j] ==0:
            return
        self.area +=1
        self.grid[i][j] = 0
        self.dfs(i+1, j)
        self.dfs(i, j+1)
        self.dfs(i-1, j)
        self.dfs(i, j-1)
        return
