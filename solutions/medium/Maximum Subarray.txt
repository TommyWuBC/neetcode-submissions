// Title: Maximum Subarray
            // Difficulty: Medium
            // Language: Python3
            // Link: https://leetcode.com/problems/maximum-subarray/

class Solution:
    def maxSubArray(self, nums: list[int]) -> int:
        maximum = nums[0]
        curr = nums[0]
        for i in range(1, len(nums)):
            curr = max(curr+nums[i], nums[i])
            if curr > maximum:
                maximum = curr
        return maximum
