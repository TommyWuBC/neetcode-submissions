// Title: Reverse Integer
            // Difficulty: Medium
            // Language: Python3
            // Link: https://leetcode.com/problems/reverse-integer/

class Solution:
    def reverse(self, x: int) -> int:
        if x >=0:
            n = str(x)
            reverse = n[::-1]
            reverse = int(reverse)
        else:
            n = str(-x)
            reverse = n[::-1]
            reverse = int(reverse)
            return reverse
            if reverse > 2**31 - 1 or reverse < -2**31:
                return 0
            if reverse > 2**31 - 1 or reverse < -2**31:
                return 0
            return -reverse
