// Title: Sort List
            // Difficulty: Medium

# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def sortList(self, head: ListNode | None) -> ListNode | None:
        arr = []
        curr = head
        while curr is not None:
            arr.append(curr)
            curr =curr.next
        arr.sort(key=lambda x: x.val)
        head = arr[0]
        curr = head
        for i in range(1, len(arr)):
            curr.next = arr[i]
            curr = curr.next
        curr.next = None
        if head is None:
            return
        return head
